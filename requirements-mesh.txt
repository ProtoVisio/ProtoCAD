# Построитель сеток для подготовки к расчёту (protocad/prep/mesh.py).
#
# gmsh распространяется под GPL-2+ и потому живёт ОТДЕЛЬНО от рабочего
# окружения ProtoCAD: ProtoCAD его не импортирует, а запускает отдельным
# процессом и обменивается файлами (так же с gmsh работает FreeCAD FEM).
# Заодно падение построителя на плохой геометрии не роняет окно.
#
# Поставить в отдельное окружение и указать его питон:
#
#   python -m venv .venv-mesh
#   .venv-mesh/bin/python -m pip install -r requirements-mesh.txt
#   set PROTOCAD_GMSH_PYTHON=.venv-mesh/bin/python   (или export на Linux)
#
# Без переменной ProtoCAD пробует свой же питон: для разработки можно
# поставить gmsh и туда, но тогда он попадает в окружение поставки.
gmsh>=4.11
numpy>=1.26
